Приклад коду

                            const ANIMATION_ATTR = 'data-animation';

const observer = new IntersectionObserver((entries) => {
    for (const entry of entries) {
        if (!entry.isIntersecting) continue;

        entry.target.setAttribute(ANIMATION_ATTR, 'visible');
        unobserveElement(entry.target);
    }
}, { threshold: 0.03 });

initElements(document.body);

function initElements(rootElement) {
    for (const element of rootElement.querySelectorAll(`[${ANIMATION_ATTR}]`))
        observeElement(element);
}

function observeElement(element) {
    observer.observe(element);
}

function unobserveElement(element) {
    observer.unobserve(element);
}

export { initElements, observeElement, unobserveElement };
